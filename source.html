src/records.mjs · lines 179–191

179  export function changeOrderStatus(db, orderId, requested, staffId) {
180    transaction(db, () => {
181      const order = db.prepare('SELECT status FROM orders WHERE id = ?').get(orderId);
182      if (!order) throw new StoreError(404, 'This order could not be found.');
183      const next = nextStatus(order.status);
184      if (!next || requested !== next) {
185        throw new StoreError(409, 'That status change is unavailable. Reopen the order to see the next step.');
186      }
187      db.prepare('UPDATE orders SET status = ? WHERE id = ?').run(next, orderId);
188      db.prepare('INSERT INTO order_events (order_id, status, actor_id, occurred_at) VALUES (?, ?, ?, ?)')
189        .run(orderId, next, staffId, new Date().toISOString());
190    });
191  }
